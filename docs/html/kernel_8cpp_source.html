130void terminal_initialize(void)
131{
132 terminal_row = 0;
133 terminal_column = 0;
134 terminal_color = VGA_COLOR_LIGHT_GREY | (VGA_COLOR_BLACK << 4);
135 terminal_buffer = VGA_MEMORY;
136 for (size_t y = 0; y < VGA_HEIGHT; y++)
137 {
138 for (size_t x = 0; x < VGA_WIDTH; x++)
139 {
140 const size_t index = y * VGA_WIDTH + x;
141 // забиваем всё пробелами
142 terminal_buffer[index] = (uint16_t)' ' | (uint16_t)terminal_color << 8;
143 }
144 }
145}

151void terminal_scroll()
152{
153 for (size_t y = 0; y < VGA_HEIGHT - 1; y++)
154 {
155 for (size_t x = 0; x < VGA_WIDTH; x++)
156 {
157 terminal_buffer[y * VGA_WIDTH + x] = terminal_buffer[(y + 1) * VGA_WIDTH + x];
158 }
159 }
160 // последнюю строчку чистим
161 for (size_t x = 0; x < VGA_WIDTH; x++)
162 {
163 terminal_buffer[(VGA_HEIGHT - 1) * VGA_WIDTH + x] = (uint16_t)' ' | (uint16_t)terminal_color << 8;
164 }
165 terminal_row = VGA_HEIGHT - 1;
166}

181void terminal_putchar(char c)
182{
183 if (c == '\n') // если встретили энтер, прыгаем на новую строку
184 {
185 terminal_column = 0;
186 if (++terminal_row == VGA_HEIGHT)
187 terminal_scroll();
188 return;
189 }
190
191 // рисуем символ
192 terminal_putentryat(c, terminal_color, terminal_column, terminal_row);
193
194 // двигаем координату вправо
195 if (++terminal_column == VGA_WIDTH)
196 {
197 terminal_column = 0;
198 if (++terminal_row == VGA_HEIGHT)
199 terminal_scroll();
200 }
201
205 uint16_t pos = terminal_row * VGA_WIDTH + terminal_column;
206 outb(0x3D4, 0x0F);
207 outb(0x3D5, (uint8_t)(pos & 0xFF));
208 outb(0x3D4, 0x0E);
209 outb(0x3D5, (uint8_t)((pos >> 8) & 0xFF));
210}

234void terminal_backspace()
235{
236 if (terminal_column == 0 && terminal_row > 0)
237 {
238 terminal_row--;
239 terminal_column = VGA_WIDTH;
240 }
241 if (terminal_column > 0)
242 {
243 terminal_column--;
244 terminal_putentryat(' ', terminal_color, terminal_column, terminal_row);
245
246 // не забываем вернуть мигающий курсор назад
247 uint16_t pos = terminal_row * VGA_WIDTH + terminal_column;
248 outb(0x3D4, 0x0F);
249 outb(0x3D5, (uint8_t)(pos & 0xFF));
250 outb(0x3D4, 0x0E);
251 outb(0x3D5, (uint8_t)((pos >> 8) & 0xFF));
252 }
253}

270void cmd_reboot()
271{
272 uint8_t temp;
273 asm volatile("cli"); // вырубаем прерывания, чтобы ничего не мешало
274 do
275 {
276 temp = inb(0x64); // ждем, пока контроллер освободится
277 if (temp & 1)
278 inb(0x60);
279 } while (temp & 2);
280 outb(0x64, 0xFE); // шлем команду на ресет
281 asm volatile("hlt");
282}

324void execute_command()
325{
326 terminal_writeln("");

329

344 terminal_writeln(" ( o.o ) ");
345 terminal_writeln(" > ^ < ");
346 terminal_writeln(" / ^ \\ ");
347 terminal_color = VGA_COLOR_LIGHT_GREY;
348 }

353 terminal_writeln(" ascii - Show art");
354 terminal_writeln(" clear - Clear screen");

356 terminal_writeln(" led 0 - Turn OFF keyboard LEDs");
357 terminal_writeln(" reboot - Restart system");
358 terminal_writeln(" power 0 - Shutdown (QEMU)");
359 }
360 else if (strcmp(cmd_buffer, "reboot"))
361 {
362 terminal_writeln("Rebooting...");
363 cmd_reboot();
364 }
365 else if (strcmp(cmd_buffer, "power 0"))
366 {
367 terminal_writeln("Shutting down...");
368 cmd_poweroff();
369 }
370 else if (strcmp(cmd_buffer, "led 1"))
371 {
372 cmd_led(1);
373 terminal_writeln("LEDs ON");
374 }
375 else if (strcmp(cmd_buffer, "led 0"))
376 {
377 cmd_led(0);
378 terminal_writeln("LEDs OFF");
379 }
380 else if (strcmp(cmd_buffer, "clear"))
381 {
382 terminal_initialize();
383 }
384 else
385 {
386 terminal_color = VGA_COLOR_LIGHT_RED;
387 terminal_write("Unknown command: ");
388 terminal_writeln(cmd_buffer);
389 terminal_color = VGA_COLOR_LIGHT_GREY;
390 }
391
392 // после выполнения чистим буфер для новой команды
393 for (int i = 0; i < 128; i++)
394 cmd_buffer[i] = 0;
395 cmd_len = 0;
396 terminal_write("> ");
397}

403extern "C" void kernel_main(void)
404{
405 // стартуем экран
406 terminal_initialize();
407
408 // приветствие
409 terminal_color = VGA_COLOR_WHITE;
410 terminal_writeln("VST Operating System");
411 terminal_writeln("Type 'help' for commands.");
412 terminal_writeln("------------------------------");
413 terminal_color = VGA_COLOR_LIGHT_GREY;
414 terminal_write("> ");
415
416 // основной цикл, тут и живет система
417 while (1)
418 {
419 // опрашиваем порт клавиатуры на нажатие чего либо
420 if (inb(0x64) & 1)
421 {
422 uint8_t scancode = inb(0x60);
423
424 // если старший бит 1, значит клавишу отпустили, нам это пока не интересно
425 if (scancode & 0x80)
426 {
427 }
428 else
429 {
430 // получаем символ из нашей таблицы
431 char c = kbd_US[scancode];
432
433 if (c == '\n') // нажали энтер - выполняем
434 {
435 execute_command();
436 }
437 else if (c == '\b') // нажали бакспейс - стираем
438 {
439 if (cmd_len > 0)
440 {
441 terminal_backspace();
442 cmd_buffer[--cmd_len] = 0;
443 }
444 }
445 else if (c > 0) // обычная буква - пишем в буфер и на экран
446 {
447 if (cmd_len < 79)
448 {
449 terminal_putchar(c);
450 cmd_buffer[cmd_len++] = c;
451 cmd_buffer[cmd_len] = 0;
452 }
453 }
454 }
455 }
